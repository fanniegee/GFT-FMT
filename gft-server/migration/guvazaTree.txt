// Admin edit authority for second surname
const adminCanEditSecondSurname = true;
function updateSecondSurname(systemId, newSecondSurname, isAdmin){
  if(!isAdmin &&!adminCanEditSecondSurname) throw "Admin only";
  if(guvazaTreeById[systemId]){
    guvazaTreeById[systemId].secondSurname = newSecondSurname;
    guvazaTreeById[systemId].secondSurnameHistory = guvazaTreeById[systemId].secondSurnameHistory || [];
    guvazaTreeById[systemId].secondSurnameHistory.push(newSecondSurname);
  }
}

// FIXED RULE: Parent determines Generation - namesake safe, delayed registration safe
function registerNewMember({fullName, parentId, dob = "", wifeNo = null}) {
  const parent = guvazaTreeById[parentId];
  if (!parent) throw "Parent ID not verified. Select parent from 224 first.";

  const childGen = parent.generation + 1;
  const branch = parent.branch;

  // Namesake check: if same name exists under different parent, it is still NEW child (don't reuse old ID)
  // Only return existing if SAME name + SAME parent + SAME DOB (returning user)
  const existingSamePerson = guvazaTree.find(m => m.fullName === fullName && m.parentId === parentId && dob && m.dob === dob);
  if (existingSamePerson) return existingSamePerson;

  // Generate next ID in that GEN (global counter for that GEN)
  const maxInGen = guvazaTree.filter(m => m.generation === childGen).length;
  const nextNum = String(maxInGen + 1).padStart(3, '0');
  const newSystemId = `GFT-GEN${childGen}-${nextNum}`;

  const newMember = {
    fullName: fullName,
    rootSurname: "Guvaza",
    currentSurname: fullName.split(" ").pop(),
    secondSurname: "",
    systemId: newSystemId,
    generation: childGen,
    branch: branch,
    parentId: parentId,
    isPlaceholder: false,
    isEditable: true,
    isFemaleChild: false,
    wifeNo: wifeNo,
    dob: dob
  };

  guvazaTree.push(newMember);
  guvazaTreeById[newSystemId] = newMember;
  return newMember;
}
